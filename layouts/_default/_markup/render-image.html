{{- define "RenderImageSimple" -}} {{- $imgObj := .imgObj -}} {{- end -}} {{- define "RenderImageResponsive" -}} {{- $imgObj := .imgObj -}} {{- $originalWidth := $imgObj.Width -}} {{- $img800 := $imgObj -}} {{- $img1280 := $imgObj -}} {{- if gt $originalWidth 800 -}} {{- $img800 = $imgObj.Resize "800x" -}} {{- end -}} {{- if gt $originalWidth 1280 -}} {{- $img1280 = $imgObj.Resize "1280x" -}} {{- end -}} {{- end -}} {{- $disableImageOptimization := .Page.Site.Params.disableImageOptimizationMD | default false -}} {{- $urlStr := .Destination | safeURL -}} {{- $url := urls.Parse $urlStr -}} {{- $altText := .Text -}} {{- $caption := .Title -}} {{- $isRemote := findRE "^(https?|data)" $url.Scheme -}} {{- $resource := "" -}} {{- $staticURL := $urlStr -}} {{- if not $isRemote -}} {{- $resourcePath := strings.TrimPrefix "/" $url.Path -}} {{- $resource = or (.Page.Resources.GetMatch $urlStr) (resources.Get $resourcePath) -}} {{- if and $resource (not (reflect.IsImageResource $resource)) -}} {{- $resource = "" -}} {{- end -}} {{- if not $resource -}} {{- $staticURL = strings.TrimPrefix "/" $urlStr | relURL -}} {{- end -}} {{- end -}}
